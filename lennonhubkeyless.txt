-- File configuration for saving key
local fileName = "LennonHub_KeyData.json"
local correctKey = "LennonHubKeyGskeo - Nskaoowheneidindjeib"

-- Function to check saved key data (24 hours validity)
local function checkSavedKey()
    if readfile and isfile and isfile(fileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(fileName))
        end)
        if success and data and data.key == correctKey then
            -- Check if 24 hours (86400 seconds) have not passed yet
            if os.time() - data.time < 86400 then
                return true
            end
        end
    end
    return false
end

-- Function to save key with current timestamp
local function saveKeyData()
    if writefile then
        local data = {
            key = correctKey,
            time = os.time()
        }
        pcall(function()
            writefile(fileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- If key is already valid within 24 hours, bypass the UI and load the script directly
if checkSavedKey() then
    loadstring(game:HttpGet("https://api.luarmor.net/files/v4/loaders/4595fe31a5f7a8b4f4dd7071f3119ef7.lua"))()
    return
end

-- Function to shake and clear text simultaneously
local function shakeAndClear()
    local originalPos = UDim2.new(0.5, -175, 0.5, -115)
    local intensity = 8
    
    TextBox.Text = ""
    
    for i = 1, 3 do
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, intensity, 0, 0)}):Play()
        task.wait(0.04)
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, -intensity, 0, 0)}):Play()
        task.wait(0.04)
    end
    TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos}):Play()
end
